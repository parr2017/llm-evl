一、项目目标

本节课从零构建一个完整的Code Review Crew——三个 Agent 协作完成代码评审：

Agent角色职责
Scanner安全审查员扫描代码中的安全漏洞和潜在风险
Architect架构师评估代码结构、设计模式和可维护性
Reviewer主评审员综合两份报告，生成最终评审意见和评分

架构：Orchestrator 模式——一个主节点负责调度，两个 Worker 并行评审，最后汇总。

# tools/code_tools.py
from langchain_core.tools import tool

@tool
def extract_code_summary(code: str) -> str:
    """提取代码的关键信息摘要，包括：函数列表、类定义、依赖导入。"""
    lines = code.strip().split("\\n")
    functions = [l.strip() for l in lines if l.strip().startswith("def ") or l.strip().startswith("async def ")]
    classes = [l.strip() for l in lines if l.strip().startswith("class ")]
    imports = [l.strip() for l in lines if l.strip().startswith("import ") or l.strip().startswith("from ")]
    
    summary = f"代码行数: {len(lines)}\\n"
    summary += f"函数 ({len(functions)}): {' | '.join(functions[:5])}" + (f" ... 等{len(functions)}个" if len(functions) > 5 else "")
    summary += f"\\n类 ({len(classes)}): {' | '.join(classes[:3])}" if classes else ""
    summary += f"\\n导入 ({len(imports)}): {' | '.join(imports[:5])}" if imports else ""
    return summary

@tool
def save_report(content: str, filename: str) -> str:
    """将评审报告保存到本地 Markdown 文件。"""
    with open(filename, "w", encoding="utf-8") as f:
        f.write(content)
    return f"✅ 报告已保存至: {filename}"

四、Scanner Agent

# agents/scanner.py
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from typing import TypedDict, Annotated
import operator

class ScannerState(TypedDict):
    code: str
    security_findings: Annotated[str, operator.add]
    risk_level: str

def scanner_node(state: ScannerState) -> dict:
    llm = ChatOpenAI(model="gpt-4o-mini")
    
    response = llm.invoke([
        ("system", """你是安全审查专家。请扫描代码中的安全问题，包括：
- 硬编码密钥/密码
- SQL注入风险
- XSS 漏洞
- 不安全的反序列化
- 权限控制缺失

对每个问题注明风险等级（high/medium/low），并给出修复建议。
如果代码没有安全问题，输出：无安全问题发现。"""),
        ("user", f"请审查以下代码的安全问题：\n\n{state['code']}")
    ])
    
    content = response.content.strip()
    risk_level = "high" if "high" in content.lower() else ("medium" if "medium" in content.lower() else "low")
    
    return {"security_findings": f"【安全审查】\\n{content}", "risk_level": risk_level}

五、Architect Agent

# agents/architect.py
from langchain_openai import ChatOpenAI
from typing import TypedDict, Annotated
import operator

class ArchitectState(TypedDict):
    code: str
    architecture_review: Annotated[str, operator.add]
    maintainability_score: int

def architect_node(state: ArchitectState) -> dict:
    llm = ChatOpenAI(model="gpt-4o-mini")
    
    response = llm.invoke([
        ("system", """你是资深软件架构师。请评估代码的可维护性，包括：
- 代码结构和模块化程度
- 命名规范和可读性
- 重复代码和冗余逻辑
- 错误处理方式
- 测试友好度

给出 1-10 的总分，并说明主要优点和改进建议。"""),
        ("user", f"请评审以下代码的架构质量：\n\n{state['code']}")
    ])
    
    content = response.content.strip()
    # 尝试从回复中提取分数
    import re
    score_match = re.search(r'[1-9]0?/10|[1-9]0?\s*分', content)
    score = 7  # 默认
    if score_match:
        try:
            score = int(re.search(r'\d+', score_match.group()).group())
        except:
            pass
    
    return {"architecture_review": f"【架构评审】\\n{content}", "maintainability_score": score}

六、Reviewer Agent（综合评审）

# agents/reviewer.py
from langchain_openai import ChatOpenAI
from typing import TypedDict

class FinalState(TypedDict):
    code: str
    security_findings: str
    architecture_review: str
    final_report: str
    overall_score: int

def reviewer_node(state: FinalState) -> dict:
    llm = ChatOpenAI(model="gpt-4o-mini")
    
    response = llm.invoke([
        ("system", """你是主评审员。请综合安全审查和架构评审报告，
生成一份结构化的最终评审报告，包括：
1. 总体评分（0-100）
2. 必须修复的问题（blocking）
3. 建议改进的问题（suggestion）
4. 亮点（positives）
5. 合入建议（merge / changes / reject）"""),
        ("user", f"""请生成最终评审报告。

【原始代码】
{state['code']}

【安全审查结果】
{state['security_findings']}

【架构评审结果】
{state['architecture_review']}""")
    ])
    
    content = response.content.strip()
    
    # 简单解析分数
    import re
    score_match = re.search(r'\d{1,3}', content)
    overall_score = 70
    if score_match:
        try:
            overall_score = min(100, int(score_match.group()))
        except:
            pass
    
    return {"final_report": content, "overall_score": overall_score}

七、工作流组装

# graph.py
from langgraph.graph import StateGraph, START, END
from typing import TypedDict
import operator

class CrewState(TypedDict):
    code: str
    security_findings: str
    architecture_review: str
    final_report: str
    overall_score: int

from agents.scanner import scanner_node
from agents.architect import architect_node
from agents.reviewer import reviewer_node

workflow = StateGraph(CrewState)

# 添加节点
workflow.add_node("scanner", scanner_node)
workflow.add_node("architect", architect_node)
workflow.add_node("reviewer", reviewer_node)

# Orchestrator 模式：scanner 和 architect 并行，reviewer 汇总
workflow.add_edge(START, "scanner")
workflow.add_edge(START, "architect")  # 两条边从 START 出发 = 并行执行

workflow.add_edge("scanner", "reviewer")
workflow.add_edge("architect", "reviewer")
workflow.add_edge("reviewer", END)

app = workflow.compile()
💡 并行执行的关键
当两个节点都从同一个节点（如 START）接收边时，LangGraph 会并行执行它们。scanner 和 architect 同时运行，互不等待，最后在 reviewer 节点汇合。

八、入口文件

# main.py
import asyncio
from graph import app
from tools.code_tools import save_report

SAMPLE_CODE = '''
import hashlib
import os

def login(username, password):
    # 硬编码密钥 - 安全隐患！
    secret_key = "my_super_secret_key_123"
    
    # SQL注入风险
    query = f"SELECT * FROM users WHERE name='{username}' AND pass='{password}'"
    
    # 不安全的哈希
    token = hashlib.md5((username + secret_key).encode()).hexdigest()
    return {"token": token, "query": query}

def process_data(raw_data):
    # 好的：输入验证
    if not raw_data or not isinstance(raw_data, dict):
        raise ValueError("Invalid input")
    
    # 好的：使用参数化查询思路（虽然这里没连接数据库）
    cleaned = {k: str(v).strip() for k, v in raw_data.items()}
    return cleaned
'''

async def main():
    print("🔍 开始 Code Review...\\n")
    
    result = await app.ainvoke({
        "code": SAMPLE_CODE,
        "security_findings": "",
        "architecture_review": "",
        "final_report": "",
        "overall_score": 0
    })
    
    print(f"📊 综合评分: {result['overall_score']}/100\\n")
    print("═" * 50)
    print(result["final_report"])
    print("═" * 50)
    
    # 保存报告
    save_report.invoke({"content": result["final_report"], "filename": "review_report.md"})

if __name__ == "__main__":
    asyncio.run(main())
